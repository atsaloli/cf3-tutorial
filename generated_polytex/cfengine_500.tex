\begin{codelisting}
\codecaption{500-010-Functions-0910-putting\_command\_output\_into\_variable.cf}
%= lang:cfengine3, options: "linenos": false
\begin{code}
bundle agent main
{
  vars:
      "my_result"
        string => execresult("/bin/ls /etc/motd /nosuchfile",
                             "noshell");

  reports:
      "Variable is $(my_result)";
}
\end{code}
